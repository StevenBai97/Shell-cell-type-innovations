# Python SAMAP
from samap.mapping import SAMAP
from samap.analysis import get_mapping_scores, GenePairFinder
from samalg import SAM
from samap.utils import save_samap, load_samap
import pickle
import pandas as pd

# Output prefix
outname = 'samap_oyster.Sme'

# Input h5ad files from previous preprocessing
filenames = {'Cg': 'oyster.samap.h5ad', 'Sm': 'Sme.samap.h5ad'}

# Initialize SAMAP object
sm = SAMAP(filenames, f_maps = 'maps/', save_processed = True)

# Save the raw SAMAP object
save_samap(sm, outname + '.pkl')

# Run the mapping algorithm
samap = sm.run()

# Save SAMAP results to pickle
save_samap(sm, outname + '_sm.pkl')
with open(outname + '_samap.pkl', 'wb') as fh: pickle.dump(samap, fh)

# Save UMAP coordinates of mapped cells
umap = pd.DataFrame(sm.samap.adata.obsm['X_umap'], index=list(sm.samap.adata.obs.index))
umap.to_csv(outname + '_umap.tsv', sep='\t', encoding='utf-8')

# Calculate cell type mapping scores
keys = {'Cg': 'cell_type', 'Sm': 'cell_type'}
D, MappingTable = get_mapping_scores(sm, keys)
MappingTable.to_csv(outname + '_annot.tsv', sep='\t', encoding='utf-8')

# Identify enriched gene pairs contributing to mapping
gpf = GenePairFinder(sm, keys=keys)
with open(outname + '_gpf.pkl', 'wb') as fh:
    pickle.dump(gpf, fh)

gene_pairs = gpf.find_all(thr=0.10)
gene_pairs.to_csv(outname + '_gene_pairs.tsv', sep='\t', encoding='utf-8')
